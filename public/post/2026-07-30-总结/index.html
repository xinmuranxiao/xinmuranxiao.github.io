网络编程

优雅退出

c风格

  • 捕获控制台/外部命令，控制退出
#include <csignal>
#include <thread>
#include <mutex>

bool bstop = 0;
condition_variable cond_quit;
mutex mutex_quit;

void sig_handler(int sig)
{
	if (sig == SIGINT || sig == SIGTERM)//SIGINT≈^C,SIGTERM≈q
	{
		unique_lock<mutex>lock_quit(mutex_quit);
		bstop = 1;
		cond_quit.notify_one();
	}
}

int main()
{
	try
	{
		boost::asio::io_context io_context;
		thread net_work_thread([&io_context]() {
			Server s(io_context, 10086);
			io_context.run();
			});
		signal(SIGINT, sig_handler);//出现SIGINT，调用sig_handler函数
		signal(SIGTERM, sig_handler);//出现SIGTERM，调用sig_handler函数
		while (!bstop)
		{
			unique_lock<mutex>lock_quit(mutex_quit);
			cond_quit.wait(lock_quit);//挂起这个锁
		}
		io_context.stop();
		net_work_thread.join();
	}
	catch (const std::exception&e)
	{
		cerr << "error is " << e.what() << endl;
	}
}

asio信号的方式

  • 自身注册为一个事件，然后捕获
int main()
{
	try
	{
		io_context io_context;
		signal_set signals(io_context,SIGINT,SIGTERM);
		signals.async_wait([&io_context](auto, auto) {
			io_context.stop();
			});
		Server s(io_context, 10086);
		io_context.run();
	}
	catch (const std::exception&e)
	{
		cerr << "error is " << e.what() << endl;
	}
}

loop 

use std::io;//导入
use rand::Rng;//trait
use std::cmp::Ordering;

fn main() {
    println!("guess math!");//打印宏
    let secret_number = rand::thread_rng().gen_range(1, 101);
    println!("猜一个数:{}",secret_number);
    loop {
        println!("猜一个数");
        let mut guess = String::new();
        io::stdin().read_line(&mut guess).expect("无法读取");
        let guess:u32=guess.trim().parse().expect("please number");
        match guess.cmp(&secret_number)
        {
            Ordering::Less => println!("Too small"),
            Ordering::Greater => println!("To big"),
            Ordering::Equal => println!("You win!"),
        }
    }
    
}

变量与可变性

不可变的变量

  • let value_name
let x;
  • 因为不可变，需要初始化
  • 运行x=6
  • 会报错:“cannot assign twice to immutable variable”

可变

  • let mut value_name;
let mut x;

常量

  • 不可以使用mut关键字
  • 声明常量使用const，他的类型必须标注
  • 常量可以在任何作用域进行声明，包括全局作用域
  • 常量只可以绑定到常量表达式，无法绑定调用的结果或者运行时才能算出的值
  • 在运行期间，声明作用域一直有效
  • 常量使用全大写，每个单词之间用下划线分开
  • const VALUE_NAME： type = value;
const MAX_POINTS: u32 =100_00;

影藏(覆盖)-showdown

  • 同c++，可以有多个同名变量在作用域
let x=1;
let x=x+1;
  • 同名的新变量类型可以与旧的变量不同

vue3

key管理

  • vue按照就低更新的通过v-for渲染的元素列表
  • 当数据顺序改变，不会移动DOM，而是就地更新重新渲染
  • 为了给vue一个提示，一遍它可以跟踪每个结点表示，重用和重新排序现有元素，为每个元素块加上key
<template>
    <h3>列表渲染</h3>
    <p v-for="(item,index) in names" :key = "index">{{ item }} - {{ index }}</p>

</template>

<script>
export default {
    data() {
        return{
            names:['xzt','jst','ittfs']
        }
    }
}
</script>
  • 添加index为key

事件处理

  • 使用v-on来监听事件，并触发时执行对应的js
  • v-on:click = “methodName”
  • 可处理的值为：内联事件(对应的js语句)，方法事件（函数名称）

内联事件

  • 用于简单场景
<template>
    <button @click="count++">Add 1</button>
    <p>"count is "{{ count }}</p>
</template>

<script>
export default {
    data() {
        return{
            count:0
        }
    }
}
</script>
  • 点击按钮+1

方法事件

  • 添加了函数
<template>
    <button @click="addCount">Add 1</button>
    <p>"count is "{{ count }}</p>
</template>

<script>
export default {
    data() {
        return{
            count:0
        }
    
    },
    methods:{
        addCount(){
            this.count+=1,
            console.log("click"),
            console.log(this.count)
        }
    }
}
</script>
  • 点击按钮会在控制台显示"click\n"count_val""

事件参数

  • 可以获取event对象
  • 使用的是js的原生event对象
<template>
    <button @click="addCount">Add 1</button>
    <p>"count is "{{ count }}</p>
</template>

<script>
export default {
    data() {
        return{
            count:0
        }
    },
        methods:{
        addCount(e){
            this.count++;
            e.target.innerHTML= "Add"+this.count
        }
    }
}
</script>
  • 实现按钮后面的数字，按后+1

事件传参 

<template>
    <p @click="getnamehandle(item)" v-for="(item,index) of names" :key="index">{{ item }}</p>
</template>
    
<script>
export default {
    data() {
        return{
            names:['xzt','jst','tfs']
        }
    },
    methods:{
        getnamehandle(name){
            console.log(name);
        }
    }
}
</script>
  • 实现了事件传参的，此例是点哪个名字，控制台输出什么

传递event对象

  • 使用$event传递
<template>
    <p @click="getnamehandle(item,$event)" v-for="(item,index) of names" :key="index">{{ item }}</p>
</template>
    
<script>
export default {
    data() {
        return{
            names:['xzt','jst','tfs']
        }
    },
    methods:{
        getnamehandle(name,e){
            console.log(name);
            console.log(name,e)
        }
    }
}
</script>
