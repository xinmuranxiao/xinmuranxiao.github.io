一些问题的记录

  • Q:一个socket为什么有多个Session？
  • 因为一个连接会有多个对话
  • Q:为什么socket必须要move
  • 因为socket的连接在一次连接只能有一次
  • static_cast 是 C++ 里用于显式类型转换的运算符，写法是 static_cast<目标类型>(表达式)。它表示“在编译期按已知类型信息进行转换”，适用于类型之间本来就有明确转换关系的场景，比如数值转换、空指针转换、基类/派生类指针转换、调用 explicit 构造函数等。
  • std::string::c_str() 和 std::string::data() 在 C++11 之前有本质区别：c_str() 返回以 \0 结尾的 const char*，而 data() 返回的缓冲区不保证以 \0 结尾，只能当普通字符数组用。C++11 之后两者行为统一了：data() 也返回以 \0 结尾的 const char*，所以对 std::string 来说基本可以混用。唯一需要注意的是自 C++17 起，data() 还有一个非 const 重载，返回 char*，允许你直接修改字符串内容；而 c_str() 仍然只返回 const char*。至于 std::string_view，它只有 data()，且不保证 null 结尾，也没有 c_str()。

#include <iostream>
#include <boost/asio.hpp>
#include <memory>
#include <cstring>
#include <vector>

#define MAX_LEN 1024*2
#define HEAD_LEN 4
#define HEAD_ID_LEN 4

namespace Messaging{

    //消息体的缓存类
    class MsgNode{
        public:
        //构造消息体
        MsgNode(int);
        //析构消息体
        ~MsgNode();
        //清空缓存
        void Clear();
        // 获取缓冲区指针
        char* GetBuf() const { return buf; }
        // 获取缓冲区总长度
        int GetTotalLen() const { return total_len; }
        // 获取当前读取位置
        int GetCurLen() const { return cur_len; }
        // 设置当前读取位置
        void SetCurLen(int len) { cur_len = len; }
        protected:
        char* buf;//消息体缓存
        int total_len;//缓存数据大小
        int cur_len;//当前读取到的位置
    };

    //读取类
    class RecvNode:public MsgNode{
        public:
        RecvNode(int ,int );
        private:
        //消息ID
        int msg_id;
    };

    class SendNode:public MsgNode{
        public:
        SendNode(int,int);
        private:
        //消息ID
        int msg_id;
    };

    //处理连接
    class Session:public std::enable_shared_from_this<Session>//具体读写措施类
    {
        public:
        //构造函数
        explicit Session(boost::asio::ip::tcp::socket socket);
        //外部启动函数
        void Start();

        private:
        //异步读函数
        void ReadHead();
        //读消息体函数
        void ReadBody(int,int);
        //异步写函数
        void Write(int ,const std::string&);

        //工作函数
        void Work(int,std::string&);

        //保存socket
        boost::asio::ip::tcp::socket sock;
        //读节点
        std::shared_ptr<RecvNode> recv_node;
        //写节点
        std::shared_ptr<SendNode> send_node;
        //头部长度
        static const int HEADER_LENGTH=HEAD_ID_LEN+HEAD_LEN;
    };

    //服务器主程序，分配Session
    class Server
    {
    public:
        //构造Server
        Server(boost::asio::io_context&,unsigned short);
        private:
        //分配Session
        void To_Session();

        //连接的accept
        boost::asio::ip::tcp::acceptor acceptor;
        //管理连接socket
        std::shared_ptr<boost::asio::ip::tcp::socket> sock;
        //管理Session的连接
        std::vector<std::shared_ptr<Session>> sessions;
    };

    void start(unsigned short port);

#include "messaging.h"

namespace Messaging{
    //构造函数
    MsgNode::MsgNode(int max_len){
        //计算出缓存空间
        total_len=max_len;
        //设定指针
        cur_len=0;
        //申请缓存
        buf=new char[total_len+1];
        //给最后一个空间为'\0'避免超出空间
        buf[total_len]='\0';
    }
    //析构函数
    MsgNode::~MsgNode(){
        //直接删除动态函数
        delete[] buf;
        std::cout<<"deleted buf"<<std::endl;
    }
    //清空缓存
    void MsgNode::Clear(){
        //给缓存赋值为'\0'
        ::memset(buf,'\0',total_len);
        //移动指针复位
        cur_len=0;
        std::cout<<"cleared buf"<<std::endl;
    }

    //接收节点
    RecvNode::RecvNode(int max_len,int _msg_id):MsgNode(max_len)
    {
        //获得ID
        msg_id=_msg_id;
    }

    //发送节点
    SendNode::SendNode(int max_len,int _msg_id):MsgNode(max_len)
    {
        //发送ID
        msg_id=_msg_id;
    }

    //构造Session
    Session::Session(boost::asio::ip::tcp::socket socket):sock(std::move(socket)) {}

    //Session调用的启动函数
    void Session::Start(){
        ReadHead();
    }

    //读数据
    void Session::ReadHead()
    {
        //先读取到头部的消息ID和消息长度
        recv_node=std::make_shared<RecvNode>(HEADER_LENGTH,0);
        //初始化缓存
        recv_node->Clear();
        //保活
        auto self=shared_from_this();
        //读取数据
        boost::asio::async_read(
            sock,boost::asio::buffer(recv_node->GetBuf(),recv_node->GetTotalLen()),
            [this,self](const boost::system::error_code& ec,
                        std::size_t)
                    {
                        //判断是否有错误
                        if(ec){
                            //有，输出错误并返回
                            std::cout<<"error in run , value is : "<< ec.message()<<std::endl;
                            return;
                        }
                        //获取读缓存指针
                        char* header =recv_node->GetBuf();
                        //获取信息ID和信息长度
                        int msg_id=0;
                        int msg_len=0;

                        //将信息ID和信息长度，读出缓存
                        std::memcpy(&msg_id,header,4);
                        std::memcpy(&msg_len,header+4,4);
                        //打印提示
                        std::cout<<"recviced message ,ID: "<<msg_id<<" and len : "<<msg_len<<std::endl;
                        //调用读信息函数
                        ReadBody(msg_id,msg_len);
                    }
        );
    }

    //异步读消息体
    void Session::ReadBody(int msg_id,int msg_len)
    {
        //创建读缓存
        recv_node=std::make_shared<RecvNode>(msg_len,msg_id);
        //清空创建缓存
        recv_node->Clear();
        auto self=shared_from_this();
        boost::asio::async_read(
            sock,boost::asio::buffer(recv_node->GetBuf(),recv_node->GetTotalLen()),
            [this,self,msg_id](const boost::system::error_code& ec,
                                std::size_t)
                                {
                                //判断是否有错误
                                if(ec){
                                //有，输出错误并返回
                                std::cout<<"error in run , value is : "<< ec.message()<<std::endl;
                                    return;
                                    }
                                //设置读的长度
                                recv_node->SetCurLen(recv_node->GetTotalLen());
                                //获得消息
                                std::string msg(recv_node->GetBuf(),recv_node->GetCurLen());
                                std::cout<<"recivced id : "<<msg_id<<" message is :"<<msg<<std::endl;
                                //调用工作函数
                                Work(msg_id,msg);
                            }
        );
    }

    //工作函数
    void Session::Work(int msg_id,std::string& msg)
    {
        std::string str="recivced message";
        Write(msg_id,str);
    }

    //异步写函数
    void Session::Write(int msg_id, const std::string& msg)
    {
        //计算发送缓存大小
        int total_len=HEADER_LENGTH+static_cast<int>(msg.size());
        //创建发送换存
        send_node=std::make_shared<SendNode>(total_len,msg_id);
        //获取创建的缓存指针
        char* buf =send_node->GetBuf();
        //获取消息体长度
        int value_len=static_cast<int>(msg.size());
        //将msg_id拷贝进缓存
        std::memcpy(buf,&msg_id,HEAD_ID_LEN);
        //将msg_len拷贝进缓存
        std::memcpy(buf+HEAD_ID_LEN,&value_len,HEAD_LEN);
        //将消息拷贝进缓存
        std::memcpy(buf+HEADER_LENGTH,msg.data(),value_len);
        //设置发送体的长度
        send_node->SetCurLen(total_len);
        //保活
        auto self=shared_from_this();
        //异步发送
        boost::asio::async_write(
            sock,
            boost::asio::buffer(send_node->GetBuf(),send_node->GetTotalLen()),
            [this,self](const boost::system::error_code& ec,std::size_t)
            {
                if(!ec)
                {
                    std::cout<<"sended message"<<std::endl;
                    ReadHead();
                }
            }
        );
    }

    //初始化Server
    Server::Server(boost::asio::io_context& ioc,unsigned short port)
    :acceptor(ioc,boost::asio::ip::tcp::endpoint(boost::asio::ip::tcp::v4(),port)){
        To_Session();
    }

    //为连接创建Session
    void Server::To_Session(){
        //异步读
        acceptor.async_accept(
            [this](const boost::system::error_code& ec, boost::asio::ip::tcp::socket socket) {
            if (!ec) {
                std::cout << "客户端已连接\n";
                auto session = std::make_shared<Session>(std::move(socket));
                sessions.push_back(session);
                session->Start();
            }
            //继续接受下一个连接
            To_Session();
            });
    }

    void start(unsigned short port){
        boost::asio::io_context io;
        Server server(io, port);
        io.run();
    }
}
